import cv2 import mediapipe as mp import pyautogui import numpy as np import time # Tắt failsafe của pyautogui (tránh dừng đột ngột) pyautogui.FAILSAFE = False pyautogui.PAUSE = 0 # Cấu hình SMOOTHING = 0.35 # Càng nhỏ càng mượt (0.2 ~ 0.5) CLICK_THRESHOLD = 0.045 # Khoảng cách chụm ngón để click SCREEN_W, SCREEN_H = pyautogui.size() mp_hands = mp.solutions.hands mp_draw = mp.solutions.drawing_utils hands = mp_hands.Hands( max_num_hands=1, # Đổi thành 2 nếu muốn dùng 2 tay model_complexity=1, min_detection_confidence=0.7, min_tracking_confidence=0.7 ) cap = cv2.VideoCapture(0) cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640) cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 480) prev_x, prev_y = 0, 0 click_pressed = False last_click_time = 0 print("Chuột ảo đã sẵn sàng!") print("Đưa tay vào camera...") print("Chụm ngón cái + ngón trỏ để click") print("Nhấn ESC để thoát") while True: success, img = cap.read() if not success: break img = cv2.flip(img, 1) # Lật gương img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB) results = hands.process(img_rgb) h, w, _ = img.shape if results.multi_hand_landmarks: for hand_landmarks in results.multi_hand_landmarks: # Vẽ khung tay mp_draw.draw_landmarks(img, hand_landmarks, mp_hands.HAND_CONNECTIONS) # Lấy tọa độ ngón trỏ (landmark 8) và ngón cái (landmark 4) index_tip = hand_landmarks.landmark[8] thumb_tip = hand_landmarks.landmark[4] # Chuyển sang tọa độ pixel trên camera cx, cy = int(index_tip.x * w), int(index_tip.y * h) # Chuyển sang tọa độ màn hình screen_x = np.interp(index_tip.x, [0.1, 0.9], [0, SCREEN_W]) screen_y = np.interp(index_tip.y, [0.1, 0.9], [0, SCREEN_H]) # Làm mượt curr_x = prev_x + (screen_x - prev_x) * SMOOTHING curr_y = prev_y + (screen_y - prev_y) * SMOOTHING # Di chuyển chuột thật pyautogui.moveTo(curr_x, curr_y) prev_x, prev_y = curr_x, curr_y # Tính khoảng cách giữa ngón cái và ngón trỏ distance = np.hypot(index_tip.x - thumb_tip.x, index_tip.y - thumb_tip.y) # Click if distance < CLICK_THRESHOLD: if not click_pressed: current_time = time.time() if current_time - last_click_time > 0.35: # chống click liên tục pyautogui.click() last_click_time = current_time click_pressed = True cv2.circle(img, (cx, cy), 15, (0, 255, 0), cv2.FILLED) else: cv2.circle(img, (cx, cy), 15, (0, 255, 0), cv2.FILLED) else: click_pressed = False cv2.circle(img, (cx, cy), 12, (255, 0, 255), cv2.FILLED) # Hiển thị cv2.putText(img, "Hand Mouse - Nhan ESC de thoat", (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 255), 2) cv2.imshow("Chuot ao bang tay", img) if cv2.waitKey(1) & 0xFF == 27: # ESC break cap.release() cv2.destroyAllWindows()
